# Define steps to safely finish the observation and leave it ready for the next observation. e.g. move to a safe position, or park. Could turn off camera etc.
# Will run even if there is an error. 
DEF SUB safe_finish
        SET GAIN TO 90
        SET RESOLUTION TO 1936x1096
        MOUNT GOTO ALTAZ 30 270
        MOUNT TRACKING None
END SUB    

SEQUENCE
    #Asteroid {object_name} at {event_time}
    #Star Mag: {star_mag};	Combined Mag: {comb_mag}; Mag Drop: {mag_drop}
    #Predict Time {event_time}; Error In Time: {time_error}
    #MaxDuration: %MaxDuration%; Probability: %Probability%

    
# Customise these to your setup
# Focal length in mm for platesolving
    SET FOCALLEN TO 660
# Your preferred output format     
    SET OUTPUT FORMAT TO "ADV files (*.adv)"     
# Full screen or resolution to use for plate solving
    SET RESOLUTION TO 1936x1096
# Get the mount moving if parked
    MOUNT TRACKING Sidereal
# A gain and exposure that will be pleasant to look at and not overwork the camera whilst waiting
# You could turn the camera off, but leaving it on gives some cold protection
    SET GAIN TO 90
    SET EXPOSURE TO 0.250
    DELAY 1
    DISPLAY STRETCH AUTO
# Custom functions for UTC Countdown
# Do NOT edit or change these!
RUN PYTHON CODE "import datetime as dt; import tempfile; import os; import time;import clr;clr.AddReference('System.Windows.Forms');from System.Windows.Forms import MessageBoxButtons;"
RUN PYTHON CODE "def seconds_to_wait_utc(target_dt): return int(max(0, (target_dt - dt.datetime.utcnow()).total_seconds()));"
RUN PYTHON CODE "def do_countdown_utc(date_string, message): target_dt = dt.datetime.strptime(date_string,'%Y-%m-%dT%H:%M:%S'); seconds_to_wait = seconds_to_wait_utc(target_dt); temp_file = os.path.join(tempfile.gettempdir(), 'temp_countdown.scs'); f = open(temp_file, 'w', encoding='utf-8'); f.write(f'COUNTDOWN FOR {{seconds_to_wait}} SHOWING \"{{message}} at {{date_string}} UTC\" '); f.close(); SharpCap.Sequencer.RunSequenceFile(temp_file) if seconds_to_wait >0 else None; seconds_to_wait = seconds_to_wait_utc(target_dt); result = (str(SharpCap.ShowMessageBox('Select Yes to restart Countdown. If you do not restart this event will not be recorded at the correct time','Warning: User stopped the Countdown',MessageBoxButtons.YesNo)) if seconds_to_wait > 0 else 'pass'); do_countdown_utc(date_string, message) if result == 'Yes' else None;"


# Countdown  to GOTO 
RUN PYTHON CODE "do_countdown_utc('{goto_time}','Waiting for GOTO')"

    # GOTO and PLATESOLVE. If an error it should run SUB safe_finish to move the mount to its defined safe position
    IGNORE ERRORS FROM ONERROR RUN safe_finish
        SET PLATESOLVE/FOCUS SETTINGS EXPOSURE 4
            SET GAIN TO 450
            SET EXPOSURE TO 0.5
	    DELAY 1
            DISPLAY STRETCH AUTO
            RETRY ERRORS UP TO 1 TIMES
                MOUNT GOTO "{ra} {dec}"
                MOUNT SOLVEANDSYNC
            END RETRY ERRORS
        END SET
    END IGNORE ERRORS


# Setup for the event
    TARGETNAME "{object_name}"

# CHange the exposure and gain to suit the event mag and duration. Change this to suit the reach of your camera and scope.
# Controlled by exposure paramete from the Occultation Manager tool
    SET RESOLUTION TO 816x822
    SET GAIN TO {gain}
    SET EXPOSURE TO {exposure}
    DELAY 2
    DISPLAY STRETCH AUTO

# Record
# Countdown to Recording start time
# If you start late it will record anyway

    RUN PYTHON CODE "do_countdown_utc('{start_time}','Waiting to record')"
    CAPTURE {recording_duration} SECONDS LIVE FRAMES
    SHOW NOTIFICATION "Recording Complete. Reseting camera and parking mount in safe position if required." COLOUR GREEN DURATION 30

## Optionally can skip if it is too late. This will only record if it is before the event mid time
#    IF EARLIER THAN LOCALTIME "{event_time_local}"
#        WAIT UNTIL LATER THAN LOCALTIME "{start_time_local}"
#        CAPTURE {recording_duration} SECONDS LIVE FRAMES
#        SHOW NOTIFICATION "Recording Complete. Reseting camera and parking mount in safe position if required." COLOUR GREEN DURATION 30
#    END IF

# run safe_finish to end    
    BEFORE SEQUENCE FINISH DO
        GOSUB safe_finish
    END DO
END SEQUENCE
