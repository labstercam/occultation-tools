# Define steps to safely finish the observation and leave it ready for the next observation. e.g. move to a safe position, or park. Could turn off camera etc.
# Will run even if there is an error. 
DEF SUB safe_finish
        SET GAIN TO 90
        SET RESOLUTION TO 1936x1096
        MOUNT GOTO ALTAZ 30 270
        MOUNT TRACKING None
END SUB

    SET RESOLUTION TO 1936x1096
# A gain and exposure that will be pleasant to look at and not overwork the camera whilst waiting
# You could turn the camera off, but leaving it on gives some cold protection
    SET GAIN TO 90
    SET EXPOSURE TO 0.250
    DELAY 1
