# SharpCap Occultation Minimal Template
# Michael Camilleri

# Example of a bare minimum template for the SharpCap OW tool automation to run
# For this to work for real, your SharpCap must be already set up with the exact camera and mount parameters required for the recording, including file format, camera gain, ROI, binning etc.
# You can use a more sophisticated template to set these up with each sequence, which is recommended

# Note that you are likely to have problems with this minimalist template as SharpCap needs to reset gain and exposure
# for plate solving, then reset it for the recording
# and it is really easy for this to get messed up resulting in a plate solve failure or a failed recording

# Will run even if there is an error. 
DEF SUB safe_finish
        # Optional
        #MOUNT GOTO ALTAZ 30 270
        #MOUNT TRACKING None
END SUB    

SEQUENCE
    #Asteroid {object_name} at {event_time}
    #Star Mag: {star_mag};	Combined Mag: {comb_mag}; Mag Drop: {mag_drop}
    #Predict Time {event_time}; Error In Time: {time_error}
    
# Get the mount moving if parked
    #MOUNT TRACKING Sidereal

    DISPLAY STRETCH AUTO
# Turn on frame timestamps
#    RUN PYTHON CODE "SharpCap.SelectedCamera.Controls.TimestampFrames.Value = True;"

# Wait until the GOTO time, which is 4 mins before event time plus 3 times the time error
# Do not change anything in this block.
    RUN PYTHON CODE "import datetime; event_date_utc = datetime.strptime('{event_time}', '%Y-%m-%dT%H:%M:%S').date(); "
    RUN PYTHON CODE "start_time = datetime.strptime('{start_time}', '%Y-%m-%dT%H:%M:%S'); "
    RUN PYTHON CODE "goto_time = datetime.strptime('{goto_time}', '%Y-%m-%dT%H:%M:%S'); "
    RUN PYTHON CODE "goto_seconds =  int(max(0, (goto_time - datetime.datetime.utcnow()).total_seconds()));"
    RUN PYTHON CODE "SharpCap.ShowNotification(f'Waiting to perform GOTO to {asteroid_name} at {goto_time},  before {event_time}.',visibleForSeconds = goto_seconds); "
    RUN PYTHON CODE "import time; time.sleep(goto_seconds); "

# GOTO and PLATESOLVE. If an error it will run SUB safe_finish, for example to move the mount to its defined safe position
IGNORE ERRORS FROM ONERROR RUN safe_finish
    SET PLATESOLVE/FOCUS SETTINGS EXPOSURE 4
	DELAY 1
        DISPLAY STRETCH AUTO
        RETRY ERRORS UP TO 1 TIMES
            MOUNT GOTO "{ra} {dec}"
            MOUNT SOLVEANDSYNC
        END RETRY ERRORS
    END SET
END IGNORE ERRORS

# Setup for the event
# Names the Sharcap target as, say 20250102 (1) Ceres. The actual file name and folder are set by your Sharpcap settings
    RUN PYTHON CODE "SharpCap.TargetName = event_date_utc.strftime('%Y%m%d') + ' {asteroid_name}';"

#    SET GAIN TO 450
    SET EXPOSURE TO {exposure}

    DELAY 2
    DISPLAY STRETCH AUTO

# Record
# Recording duration - default is 60s + event duration (if > 5 s) + 6x timeerror if > 2s error
# Do not change any code in this block.
    RUN PYTHON CODE "from SharpCap.UI import CaptureLimitType; SharpCap.SelectedCamera.CaptureConfig.CaptureLimitType = CaptureLimitType.TimeLimited;"
    RUN PYTHON CODE "import System, datetime; SharpCap.SelectedCamera.CaptureConfig.CaptureLimitTime = System.TimeSpan.FromSeconds({recording_duration});"
    RUN PYTHON CODE "import datetime; record_seconds = int(max(0,(start_time - datetime.datetime.now()).total_seconds());"
    RUN PYTHON CODE "SharpCap.ShowNotification('Waiting to record at {start_time} for event at {event_time}',visibleForSeconds = record_seconds); "
    RUN PYTHON CODE "import time; time.sleep(record_seconds);"
    RUN PYTHON CODE "SharpCap.SelectedCamera.PrepareToCapture(); SharpCap.SelectedCamera.RunCapture();"
    RUN PYTHON CODE "SharpCap.ShowNotification('Recording {asteroid_name} for {recording_duration} seconds',visibleForSeconds = {recording_duration}; "
    RUN PYTHON CODE "import time; time.sleep({recording_duration});"
    DELAY 3
    SHOW NOTIFICATION "Recording Complete." COLOUR GREEN DURATION 30

# run safe_finish to end    
    BEFORE SEQUENCE FINISH DO
        GOSUB safe_finish
    END DO
END SEQUENCE
